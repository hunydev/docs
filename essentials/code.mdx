## 기본 REST 호출 템플릿

<CodeGroup>
```bash cURL
curl -X POST "https://api.huny.dev/v1/records" \
  -H "Authorization: Bearer $HUNY_API_KEY" \
  -H "Huny-Workspace: $HUNY_WORKSPACE" \
  -H "Idempotency-Key: evt_{{uuid}}" \
  -H "Content-Type: application/json" \
  -d @payload.json
```

## 재시도 + 에러 핸들링 패턴

```typescript
import pRetry from 'p-retry';
import { HunyClient, HunyError } from '@huny/sdk';

const client = new HunyClient({ /* ... */ });

async function sendWithRetry(payload) {
  return pRetry(async () => {
    try {
      return await client.records.create(payload);
    } catch (error) {
      if (error instanceof HunyError && error.status < 500) {
        throw new pRetry.AbortError(error.message);
      }
      throw error;
    }
  }, { retries: 3 });
}
```

## 서버리스 함수 예제

```javascript
export default async function handler(req, res) {
  const client = new HunyClient({
    apiKey: process.env.HUNY_API_KEY,
    workspaceId: process.env.HUNY_WORKSPACE,
    fetchOptions: { keepalive: false }
  });

  await client.records.create({
    streamId: 'strm_92bx12',
    payload: req.body
  });

  return res.status(200).json({ ok: true });
}
```

## Go 언어(커뮤니티 SDK)

```go
package main

import (
  "github.com/huny-labs/huny-go"
)

func main() {
  client := huny.NewClient(os.Getenv("HUNY_API_KEY"), huny.WithWorkspace(os.Getenv("HUNY_WORKSPACE")))

  _, err := client.CreateRecord(context.Background(), huny.CreateRecordInput{
    StreamID: "strm_92bx12",
    Payload: map[string]any{
      "id": "ord_123",
      "event": "checkout.completed",
    },
  })

  if err != nil {
    log.Fatal(err)
  }
}
```
